-- 1. Configuracion de Probabilidades y Contenedores
local PROBABILIDAD_SPAWN = 3 -- 3% de probabilidad de aparecer por cada contenedor valido

-- Lista de contenedores donde tiene sentido que aparezca una cinta VHS
local CONTENEDORES_VALIDOS = {
    ["bin"] = true,           -- Basureros generales
    ["crate"] = true,         -- Cajas de madera
    ["sidetable"] = true,     -- Mesitas de noche
    ["shelves"] = true,       -- Estanterias
    ["wardrobe"] = true,      -- Armarios
    ["displaycase"] = true,   -- Vitrinas (como en tiendas)
    ["desk"] = true,          -- Escritorios
    ["filingcabinet"] = true  -- Archivadores
}

-- Lista con los IDs exactos de las peliculas que registraste en RecMedia
local PELICULAS_DISPONIBLES = {
    "boondocks",
    "noticiero",
    "walkin",
    "trailer",
    "ricardo_milos",
    "joven",
    "rickroll"
}

-- =============================================================================
-- 2. Funcion Principal de Inyeccion
-- =============================================================================
local function GenerarVHSAleatorio(roomName, containerType, container)
    -- Verificamos si el tipo de contenedor esta en nuestra lista permitida
    if CONTENEDORES_VALIDOS[containerType] then
        
        -- Lanzamos un dado del 0 al 99. Si cae por debajo de nuestra probabilidad, spawnea.
        if ZombRand(100) < PROBABILIDAD_SPAWN then
            
            -- Generamos el item fisico en el contenedor
            local item = container:AddItem("Base.VHS_Animado")
            
            if item then
                -- Elegimos una pelicula al azar de nuestra tabla
                local randomIndice = ZombRand(#PELICULAS_DISPONIBLES) + 1
                local peliculaElegida = PELICULAS_DISPONIBLES[randomIndice]
                
                -- Buscamos los datos de la pelicula en la radio del juego
                local recordedMedia = getZomboidRadio():getRecordedMedia()
                local mediaData = recordedMedia:getMediaData(peliculaElegida)
                
                -- Se la asignamos al VHS recien creado
                if mediaData then
                    item:setRecordedMediaData(mediaData)
                    
                    -- Mensaje en consola para que sepas exactamente donde se genero (util para testear)
                    print(string.format("[ModVHS-Spawner] EXITO: Generado '%s' en un(a) %s dentro de la habitacion '%s'.", 
                          tostring(mediaData:getTranslatedItemDisplayName()), 
                          tostring(containerType), 
                          tostring(roomName)))
                end
            end
        end
    end
end

-- =============================================================================
-- 3. Conectar la funcion al motor del juego
-- =============================================================================
-- Este evento se dispara nativamente cada vez que el juego llena un contenedor inexplorado
Events.OnFillContainer.Add(GenerarVHSAleatorio)

-- =============================================================================

local PELICULAS_DISPONIBLES = {
    "boondocks", "noticiero", "walkin", "trailer", 
    "ricardo_milos", "joven", "rickroll"
}

-- Agrega aqui los contenedores donde quieres que aparezca
local CONTENEDORES_VALIDOS = {
    ["bin"] = true,
    ["dumpster"] = true,
    ["crate"] = true,
    ["shelves"] = true,
    ["sidetable"] = true,
    ["wardrobe"] = true,
    ["displaycase"] = true
}

-- PROBABILIDAD PRINCIPAL: ¿Que porcentaje de los contenedores validos tendran VHS?
local PROB_CONTENEDOR_PREMIADO = 100 -- 15% de probabilidad

local function GenerarVHSDinamico(roomName, containerType, container)
    -- 1. FILTRO DE CONTENEDOR
    if not CONTENEDORES_VALIDOS[containerType] then return end

    -- 2. ESCUDO ANTI-REPLICACION (Atado al mueble fisico)
    local parentObject = container:getParent()
    if not parentObject then return end

    local modData = parentObject:getModData()
    if modData.vhs_procesado_avanzado then return end
    modData.vhs_procesado_avanzado = true

    -- 3. PROBABILIDAD DE APARECER (Si el dado cae alto, no generamos nada)
    -- ZombRand(100) genera un numero del 0 al 99.
    if ZombRand(100) >= PROB_CONTENEDOR_PREMIADO then
        return -- No hubo suerte, el contenedor se queda sin nuestras cintas
    end

    -- 4. DETERMINAR CANTIDAD DE CINTAS
    local cantidad = 1
    local tiradaCantidad = ZombRand(100)
    
    if tiradaCantidad < 5 then
        -- 5% de probabilidad de JACKPOT: Aparecen TODAS las peliculas
        cantidad = #PELICULAS_DISPONIBLES
    elseif tiradaCantidad < 30 then
        -- 25% de probabilidad de encontrar un lote (2 o 3 cintas)
        cantidad = ZombRand(2, 4) -- Retorna 2 o 3
    else
        -- 70% de probabilidad de encontrar solo 1 cinta
        cantidad = 1
    end

    -- 5. SELECCION ALEATORIA SIN REPETIR
    -- Hacemos una copia temporal de la lista para ir tachando las que ya salieron
    local pelisTemp = {}
    for _, peli in ipairs(PELICULAS_DISPONIBLES) do
        table.insert(pelisTemp, peli)
    end

    local recordedMedia = getZomboidRadio():getRecordedMedia()
    
    -- Generamos la cantidad de items calculada
    for i = 1, cantidad do
        if #pelisTemp == 0 then break end -- Seguro por si acaso
        
        -- Elegir una pelicula al azar de las que quedan y sacarla de la lista temporal
        local rIndex = ZombRand(#pelisTemp) + 1
        local peliculaElegida = table.remove(pelisTemp, rIndex)

        local item = container:AddItem("Base.VHS_Animado")
        if item then
            local mediaData = recordedMedia:getMediaData(peliculaElegida)
            if mediaData then
                item:setRecordedMediaData(mediaData)
                print(string.format("[ModVHS-Spawner] Generado: %s en %s (%d/%d)", 
                      mediaData:getTranslatedItemDisplayName(), containerType, i, cantidad))
            end
        end
    end
end

Events.OnFillContainer.Add(GenerarVHSDinamico)

-- =============================================================================
-- SISTEMA DE APARICION EN COORDENADAS EXACTAS (EASTER EGGS)
-- =============================================================================

local PELICULAS_DISPONIBLES = {
    "boondocks", "noticiero", "walkin", "trailer", 
    "ricardo_milos", "joven", "rickroll"
}

-- Define aqui las coordenadas (X, Y, Z) exactas del mapa.
local COORDENADAS_ESPECIALES = {
    { x = 10613, y = 9312, z = 0 }, -- Coordenada 1 (Ejemplo)
    { x = 12500, y = 8400, z = 1 }  -- Coordenada 2 (Ejemplo)
}

local function GenerarVHSEnCoordenada(square)
    if not square then return end

    local sqX = square:getX()
    local sqY = square:getY()
    local sqZ = square:getZ()

    -- 1. VERIFICAR SI ESTAMOS EN LA COORDENADA CORRECTA
    local esCoordenadaObjetivo = false
    for _, coord in ipairs(COORDENADAS_ESPECIALES) do
        if sqX == coord.x and sqY == coord.y and sqZ == coord.z then
            esCoordenadaObjetivo = true
            break
        end
    end

    if not esCoordenadaObjetivo then return end

    -- 2. ESCUDO ANTI-REPLICACION
    local modData = square:getModData()
    if modData.vhs_coordenada_generada then return end
    modData.vhs_coordenada_generada = true

    -- 3. SELECCIONAR PELICULA ALEATORIA
    local randomIndice = ZombRand(#PELICULAS_DISPONIBLES) + 1
    local peliculaElegida = PELICULAS_DISPONIBLES[randomIndice]
    
    local recordedMedia = getZomboidRadio():getRecordedMedia()
    local mediaData = recordedMedia:getMediaData(peliculaElegida)
    
    if not mediaData then return end

    -- 4. BUSCAR UN CONTENEDOR EN ESTA CASILLA
    local targetContainer = nil
    local objects = square:getObjects()
    for i = 0, objects:size() - 1 do
        local obj = objects:get(i)
        if obj:getContainer() then
            targetContainer = obj:getContainer()
            break
        end
    end

    -- 5. INYECTAR EL ITEM
    if targetContainer then
        -- Si hay un mueble, lo escondemos adentro
        local item = targetContainer:AddItem("Base.VHS_Animado")
        if item then
            item:setRecordedMediaData(mediaData)
            print(string.format("[ModVHS-Coords] VHS Especial '%s' escondido en un mueble en (%d, %d, %d)", 
                  tostring(mediaData:getTranslatedItemDisplayName()), sqX, sqY, sqZ))
        end
    else
        -- CORRECCION: En Lua usamos instanceItem para crear de la nada
        local item = instanceItem("Base.VHS_Animado")
        if item then
            item:setRecordedMediaData(mediaData)
            square:AddWorldInventoryItem(item, 0.5, 0.5, 0.0)
            print(string.format("[ModVHS-Coords] VHS Especial '%s' tirado en el suelo en (%d, %d, %d)", 
                  tostring(mediaData:getTranslatedItemDisplayName()), sqX, sqY, sqZ))
        end
    end
end

Events.LoadGridsquare.Add(GenerarVHSEnCoordenada)
